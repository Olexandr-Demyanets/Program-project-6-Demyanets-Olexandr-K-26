# Нулі в будь-якому записі (0, -0.0, 0e5, 0.000) не враховуються в середньому,
# але кожен зупиняє сопрограму щонайменше на 1 секунду.
3 6 9
0            # пауза; середнє лишається 6
12
-0.0 0e5     # два нулі поспіль - дві окремі паузи
15
0.000        # нуль останнім: після паузи сопрограма завершується
